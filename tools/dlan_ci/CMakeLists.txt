cmake_minimum_required(VERSION 3.25)
project(uipc_dlan_ci LANGUAGES C CXX)

set(UIPC_MUDA_USE_DLAN ON CACHE BOOL "Build the actual DLAN backend" FORCE)
set(UIPC_WITH_CUDA_BACKEND ON CACHE BOOL "" FORCE)
set(UIPC_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(UIPC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UIPC_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(UIPC_BUILD_PYBIND OFF CACHE BOOL "" FORCE)
set(UIPC_BUILD_CUDA_API_PROBE OFF CACHE BOOL "" FORCE)
set(UIPC_BUILD_COREX_API_TESTS OFF CACHE BOOL "" FORCE)
set(UIPC_WITH_URDF_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. libuipc)

add_executable(dlan_physics physics.cpp)
target_link_libraries(dlan_physics PRIVATE uipc::uipc)
target_compile_definitions(dlan_physics PRIVATE
    UIPC_DLAN_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../..")
uipc_target_set_output_directory(dlan_physics)
uipc_target_add_backend_dependency(dlan_physics)

enable_testing()
foreach(mass IN ITEMS 0.01 1 1000)
    add_test(NAME dlan_free_fall_${mass} COMMAND dlan_physics free-fall ${mass})
    add_test(NAME dlan_force_${mass} COMMAND dlan_physics force ${mass})
endforeach()
add_test(NAME dlan_initial_velocity COMMAND dlan_physics initial-velocity 1)
add_test(NAME dlan_stack COMMAND dlan_physics stack 1)
get_property(dlan_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${dlan_tests} PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
